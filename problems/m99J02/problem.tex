\begin{alist}
%------------------------------------------------------------------*
\item Assume that $u_1$ and $u_2$ are twice continously differentiable
functions on $\R$ and define the function $g_y$ so that
$g_y(x)=u_1(x)u_2(y)$ if $x\leq y$ and $g_y(x)=u_2(x)u_1(y)$ if $x\geq
y$. Compute the distribution $\ddk{g_y}{x}{2}$ in $\mathcal{D}'(\R)$.
%------------------------------------------------------------------*
\item Assume in addition that $u_1$ and $u_2$ are solutions of the
equation $u''(x)+q(x)u(x)=0$. Show that
$\ddk{g_y}{x}{2}+qg_y=c\delta_y$ in $\mathcal{D}'(\R)$, where
$\delta_y(\varphi)=\varphi(y)$ and $c$ is a constant that does not
depend on $y$.
%------------------------------------------------------------------*
\end{alist}
