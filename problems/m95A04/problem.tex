\begin{alist}
%------------------------------------------------------------------*
\item Explain precisely what is meant by saying that all locally
integrable functions are distributions.
%------------------------------------------------------------------*
\item Define the derivative of a distribution.
%------------------------------------------------------------------*
\item Compute the first and second distributional derivatives of the
function $f(x)=|x|$.
%------------------------------------------------------------------*
\item Show that the derivative operator acting on the space of test
functions is injective, bet when it acts on the space of
distributions, it is not injective.
%------------------------------------------------------------------*
\item Find a distributional solution of the differential equation
$(d/dt)T+\lambda T=\delta$, where $T$ is the unknown distribution and
$\delta$ is the Dirac distribution.
%------------------------------------------------------------------*
\item Show that the space $W=\{w\in L^p([0,1])\pipe\text{\ the
distributional derivative\ } Dw\in L^p([0,1])\}$ is a Banach space
when equiped with the norm $\|w\|_{W}=[\int_{0}^1
(|u|^p+|Du|^p)dx]^{1/p}$.
%------------------------------------------------------------------*
\end{alist}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%


\soln{(b: distributional derivative) \defn{
\item  $u'(\phi)=-u(\phi')$. \\ 
\item More generally, $\partial^{\alpha} u = (-1)^{|\alpha|}u \circ
D^{\alpha}$}}
