Given a function $\varphi:\Rtw\rightarrow \Rtw$, define, whenever
possible, $T(\varphi)=\int_{-\infty}^{\infty}\varphi(s,-s)ds$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Does this formula define an element of $\mathcal{D}'$? Explain
very breifly (you don't need to prove continuity).
%------------------------------------------------------------------*
\item Does $T$ have a Fourier transform? Give a formula for it if so.
%------------------------------------------------------------------*
\item Compute $\pbyp{x}T$.
%------------------------------------------------------------------*
\item Show that $[(\pbyp{x})^2-(\pbyp{y})^2]T=0$ as a 
distribution.
%------------------------------------------------------------------*
\end{alist}
