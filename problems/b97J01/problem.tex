\begin{alist}
%------------------------------------------------------------------*
\item Define what it means for a group $G$ to be \emph{solvable} and 
\emph{nilpotent}. 
%------------------------------------------------------------------*
\item Give examples (with explanations) of a group that is not solvable
and a group that is solvable but not nilpotent.
%------------------------------------------------------------------*
\item Prove that a finite group whose order is a prime power $p^n>1$ has a non-trivial center.
%------------------------------------------------------------------*
\item Prove that a finite group of prime power order is nilpotent.
%------------------------------------------------------------------*
\end{alist}
