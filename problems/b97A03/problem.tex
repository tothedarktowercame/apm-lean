Let $f(x)=x^6+3$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that $f(x)$ is irreducible in $\Q[x]$.
%------------------------------------------------------------------*
\item Let $\alpha\in \C$ be a root of $f(x)$. Show that $f(x)$ splits
over $\Q(\alpha)$.
%------------------------------------------------------------------*
\item Compute the Galois group of $\Q(\alpha)/\Q$.
%------------------------------------------------------------------*
\item Find all subfields of $\Q(\alpha)$ and carefully show that the
list is complete.
%------------------------------------------------------------------*
\end{alist}
