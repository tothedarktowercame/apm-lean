%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Let $A$ be an arbitrary collection of open subintervals of a
bounded interval $I$ in $\R$, and let $U$ be the union of the members
of $A$. Prove that $A$ includes a sequence (possibly finite) of
pairwise disjoint intervals $I_1,I_2,\ldots $ such that $U$ is
contained in $\bigcup_jI'_j $, where $I'_j$ signifies the open
interval that is concentric with $I_j$ but five times as long.
%------------------------------------------------------------------*
\item Suppose that $I$ is an open interval $\R$ and that
$f:I\rightarrow \R$ is an arbitrary function. Let $C$ denote the set
of points $x$ in $I$ at which $f$ is differentiable and has $f'(x)=0$.
Show that $f(C)$ has Lebesgue measure zero. Hint: even if you are unable
to do part (i), you might attempt to use (i) to do (ii).
%------------------------------------------------------------------*
\end{alist}
