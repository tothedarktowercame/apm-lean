Let $\Omega=[0,1]^2\subset \Rn$ and $C^{\infty}(\Omega)=\{u\in
C^{\infty}(\Omega) \pipe u \mathrm{\ periodic}\}$. Let $H^1_P(\Omega)$
be the completion of $C^{\infty}_P(\Omega)$ in $H^1(\Omega)$ and let
$H^1_{P,0}(\Omega)=\{u\in H^1_P(\Omega) \pipe \int_{\Omega}u(x)dx=0\}$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that $H_{P,0}^1(\Omega)$ is a Hilbert space with the
$H^1(\Omega)$ inner product.
%------------------------------------------------------------------*
\item Prove that there is some constant $C>0$ such that for any $u\in
H^1_{P,0}(\Omega)$ we have $\int_{\Omega}|u(x)|^2 dx\leq
C\int_{\Omega} |\nabla u(x)|^2 dx$.
%------------------------------------------------------------------*
\item For some fixed $u_0\in H^1_p(\Omega)$ and $f\in L^2(\Omega)$,
consider the variational problem of finding $u\in
H^1_{P,0}(\Omega)+u_0$ such that $\forall v\in H^1_{P,0}(\Omega):
\int_{\Omega}\nabla u(x)\cdot \nabla v(x)dx =\int_{\Omega}f(x)v(x)dx$.
Prove that there is a unique solution.
%------------------------------------------------------------------*
\item Show that (c) is equivalent to the boundary value problem of
finding $u\in H^1_{P,0}(\Omega)+u_0$ such that $-\triangle u=f$.
%------------------------------------------------------------------*
\end{alist}
