Let $A$ be a closed subspace of a normal space $X$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that any continuous function $f:A\rightarrow \Rn$ extends to a
continuous function $F:X\rightarrow \Rn$.
%------------------------------------------------------------------*
\item Show that any continuous function $f:A\rightarrow \Sn$ extends
to a continuous function $F:U\rightarrow \Sn$ for some open
neighborhood $U$ of $A$ in $X$.
%------------------------------------------------------------------*
\end{alist}
