Let $G$ be a group of order $5075=5^2\cdot 7 \cdot 29$. Let $P$
be a Sylow $5$-subgroup of $G$, let $Q$ be a Sylow $29$-subgroup of
$G$ and let $R$ be a Sylow $7$-subgroup of $G$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that $P$ is a normal subgroup of $G$.
%------------------------------------------------------------------*
\item Show that $G$ has a normal subgroup $H$ of order $5^2\cdot
29$. Hint: Look at $G/P$.
%------------------------------------------------------------------*
\item Show that $Q$ is a normal subgroup of $G$. Hint: relate $Q$ to
$H$.
%------------------------------------------------------------------*
\item Show that $G$ has a subgroup $K$ of order $5^2\cdot 7$. 
%------------------------------------------------------------------*
\item Show that $K$ is normal if and only if $G$ is abelian.
%------------------------------------------------------------------*
\end{alist}
