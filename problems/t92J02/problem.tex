\begin{alist}
%------------------------------------------------------------------*
\item Show that the continuous image of a compact space is compact.
%------------------------------------------------------------------*
\item Suppose that $X$ is a compact space and $Y$ is Hausdorff and
$f:X\rightarrow Y$ is a continuous map that is one-to-one and
onto. Show that $f$ is a homeomorphism.
%------------------------------------------------------------------*
\end{alist}
