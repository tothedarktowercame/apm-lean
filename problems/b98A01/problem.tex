\begin{alist}
%------------------------------------------------------------------*
\item Let $G$ be a finite group and let $H$ be a subgroup of $G$ of
index $n$. Let $S_n$ be the symmetric group on $n$ elements. Show that
there is a homomorphism $\psi$ from $G$ to $S_n$ such that if $K$ is
the kernel of $\psi$ then $K\subset H$ and if $N$ is a normal subgroup
of $G$ with $N\subset H\subset G$ then $N\subset K$.
%------------------------------------------------------------------*
\item Suppose that $G$ is a group of order $210=2\cdot 3\cdot 5\cdot
7$ and let $B$ be a subgroup of $G$ of order $5$. Suppose $B$ is
\emph{not} normal in $G$. Let $N$ be the normalizer of $B$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{rlist}
%------------------------------------------------------------------*
\item Show that the order of $N$ is either $10$ or $35$.
%------------------------------------------------------------------*
\item Show that if $G$ contains a subgroup of order $10$, then the
order of $N$ is $10$.
%------------------------------------------------------------------*
\item Show that $G$ contains a subgroup of order $10$ or a subgroup of
order $35$ but \emph{not both}.
%------------------------------------------------------------------*
\item Show that $G$ contains a subgroup of order $3$ if and only if
$G$ contains a \emph{normal} subgroup of order $7$. Hint: use (a). 
%------------------------------------------------------------------*
\end{rlist}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%------------------------------------------------------------------*
\end{alist}
