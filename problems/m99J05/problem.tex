Let $V$ be a real Hilbert space, $f\in V'$ a linear functional on $V$
and define the \emph{quadratic function} $\varphi:V\rightarrow \R$ by
$\varphi(v)=\frac{1}{2}\|v\|^2-f(v)$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that there exists a unique minimum point $u\in
V:\varphi(u)\leq \varphi(v)$ for all $v\in V$.
%------------------------------------------------------------------*
\item Show that any such minimum point $u$ satisfies 
$ \forall v\in V:(u,v)_V=f(v)$.
%------------------------------------------------------------------*
\item Conversely, show that $u$ satisfying this condition is a minimum
point.
%------------------------------------------------------------------*
\end{alist}
