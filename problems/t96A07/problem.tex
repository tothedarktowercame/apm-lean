Define $f:\Stw\rightarrow \Rth$ by $f(x,y,z)=(yz,xz,xy)$. Here $\Stw$
is the set of points a distance $1$ from the origin in $\Rth$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that $f$ induces a map $\bar{f}$ of $\RP^2$ to $\Rth$. Here
$\RP^2=\Stw/\{\mathbf{x}\sim -\mathbf{x}\}$.
%------------------------------------------------------------------*
\item Decide whether or not the $z$-axis is transverse to $\bar{f}$.
%------------------------------------------------------------------*
\item For each $z_0\in \R$ let $H_{z_0}=\{(x,y,z)\pipe z=z_0\}$ be a
horizontal plane in $\Rth$. Show that there are infinitely many
horizontal planes which are transverse to $f$ with the property that
$H_{z_0}\cap
\im f\neq \phi$.
%------------------------------------------------------------------*
\end{alist}
