Let $\mathbf{Q}$ denote the field of rational numbers.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Find a Galois extension of $\mathbf{Q}$ with Galois group 
isomorphic to $\Z/3\Z$.
%-----------------------------------------------------------------*
\item Find a Galois extension of $\mathbf{Q}$ with Galois group
isomorphic to $\Z/3\Z\times \Z/3\Z$.
%------------------------------------------------------------------*
\end{alist}
