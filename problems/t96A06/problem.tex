Let $f:\Rfo\rightarrow \Rfo$ be given by $(x,y,z,w)\rightarrow
(x^2-y^2,2 xy, z^2-w^2,2zw)$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that $f$ induces a map $\bar{f}:\RP^3\rightarrow \RP^3$.
(Recall $\RP^3=(\Rfo\setminus \{\mathbf{0}\})/\{\mathbf{x}\sim
k\mathbf{x}\}$.)
%------------------------------------------------------------------*
\item Compute the Lefschetz number of $\bar{f}$.
%------------------------------------------------------------------*
\item Is $\bar{f}$ homotopic to the identity?
%------------------------------------------------------------------*
\end{alist}
