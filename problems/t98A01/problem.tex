\begin{alist}
%------------------------------------------------------------------*
\item Suppose that $\pi:\Sigma\rightarrow \Sigma'$ is a regular
$5$-sheeted covering space between closed, orientable surfaces. Is it
possible that $\Sigma$ is homeomorphic to $\Sigma'$? Construct an
example or prove it is not possible.
%------------------------------------------------------------------*
\item Let $T$ denote the $2$-torus, $P$ the projective plane, and $K$
the Klein bottle. Identify the cannonical form for the following
surface and compute its Euler characteristic: $K\# T\# T\# P\# K\# T$.
%------------------------------------------------------------------*
\end{alist}
