Suppose that $k$ is a field of characteristic $p>0$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item If $a$ is an element of $k$ show that the polynomial
$f(x)=x^p-a$ either splits in $k[x]$ or is irreducible in $k[x]$.
%------------------------------------------------------------------*
\item Let $\sigma:k\rightarrow k$ be the Frobenius endomorphism
defined by $\sigma(\alpha)=\alpha^p$. Assume that $\sigma$ is an
automorphism, \ie\ assume that $\sigma$ is surjective. Prove that
every irreducible polynomial in $k[x]$ is separable.
%------------------------------------------------------------------*
\end{alist}
