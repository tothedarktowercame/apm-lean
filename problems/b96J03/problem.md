%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Define \emph{nilpotent} and \emph{solvable} groups.
%------------------------------------------------------------------*
\item Either prove or give a counterexample to the following
statements:
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{rlist}
%------------------------------------------------------------------*
\item A nilpotent group is solvable.
%------------------------------------------------------------------*
\item A solvable group is nilpotent.
%------------------------------------------------------------------*
\end{rlist}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%------------------------------------------------------------------*
\item Let $G$ be the group with identity $e$ generated by elements
$a,b,c$ satisfying $a^2=b^3=c^7=e$, $ab=ba$ and $ac=ca$. Determine
whether $G$ is solvable and whether $G$ is nilpotent.
%------------------------------------------------------------------*
\end{alist}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\soln{(nilpotent) \defn{
\item $\exists n \in \N: r^n = 0$ }
(solvable group)  \defn{
\item $G$ group
\item $G_1=\text{commutator subgroup}(G)$
\item $\forall i\in \N: G_{i+1}=\text{commutator subgroup}(G_i)$
\item $\exists n\in \N: G_{n}=\id_G$}
(every nilpotent group is solvable) t}
