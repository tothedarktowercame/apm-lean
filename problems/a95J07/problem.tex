If $I\subset (\alpha,\beta)\subset \R$ is a bounded open interval, let
$I'=(3\alpha-2\beta,3\beta-2\alpha)$. Note that $I$ and $I'$ have the
same center but the length of $I'$ is five times the length of
$I$. Then let $A$ be a nonempty collection of open subintervals of
$(0,1)$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Prove that there is a sequence (possibly finite) of pairwise
disjoint intervals $I_1,I_2,\ldots $ in $A$ such that $\bigcup_A
I\subset \bigcup_{n=1}^{\infty}I_n'$.
%------------------------------------------------------------------*
\item Suppose that $f:(0,1)\rightarrow \R$ is an arbitrary function 
and $$D=\{x\in (0,1)\pipe f \text{\ is differentiable at\ } x,
f'(x)=0\}.$$ Show that $f(D)$ has Lebesgue measure $0$.
%------------------------------------------------------------------*
\end{alist}
