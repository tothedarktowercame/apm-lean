Let $F\subset F(\alpha)\subset E$ with $E/F$ a finite Galois
extension. Let $G$ be the Galois group of $E/F$ and let $H$ be the
Galois group of $E/F(\alpha)$. Let $\sigma_1,\ldots \sigma_n$ be a
complete set of left coset representatives for $H$ in $G$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Suppose $\sigma$ and $\tau$ are both in $G$ and let $\sigma H$
and $\tau H$ be the corresponding left cosets of $H$. Show that
$\sigma H = \tau H$ if and only if $\sigma(\alpha)=\tau(\alpha)$.
%------------------------------------------------------------------*
\item  Let $\tau$ be an arbitrary element of
$G$. Show that $\tau\sigma_1,\ldots ,\tau\sigma_{n}$ is a complete set
of left coset representatives for $H$ in $G$.
%------------------------------------------------------------------*
\item Let $g(x)=(x-\sigma_1(\alpha))(x-\sigma_2(\alpha))\cdots
(x-\sigma_n(\alpha))$. Show that $g(x)\in F[x]$.
%------------------------------------------------------------------*
\item Show that $g(x)$ as defined in part (c) is the minimal
polynomial for $\alpha$ over $F$. 
%------------------------------------------------------------------*
\end{alist}
