Let $G$ be a group and let $G'$ be the commutator subgroup of
$G$. Recall that $G'$ is the subgroup of $G$ generated by all elements
of the form $a^{-1}b^{-1}ab$ with $a$ and $b$ in $G$. Let $H$ be any
subgroup of $G$. 

A \emph{Well Known} theorem says that $G'\subset H$ if and only if $H$
is a normal subgroup of $G$ and $G/H$ is abelian.  You may use this
theorem without proving it.

In what follows, Galois extensions are assumed to be finite.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Let $F\subset E$ be a Galois extension of fields. Show that
there is a field $A$ with $F\subset A\subset E$ such that
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{rlist}
%------------------------------------------------------------------*
\item $A/F$ is Galois and
%------------------------------------------------------------------*
\item the Galois group of $A/F$ is abelian and
%------------------------------------------------------------------*
\item if $K$ is any field with $F\subset K\subset E$ with $K/F$ Galois
and with the Galois group $K/F$ abelian, then $K\subset A$. (Thus, $A$
is the unique largest field between $F$ and $E$ satisfying (i) and
(ii).)
%------------------------------------------------------------------*
\end{rlist}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%------------------------------------------------------------------*
\item With $F\subset A\subset E$ as in part (a), suppose that $K$ is a
field with $F\subset K\subset E$. Show that the following things are
equivalent
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{rlist}
%------------------------------------------------------------------*
\item $K/F$ is Galois and the Galois group of $K/F$ is abelian.
%------------------------------------------------------------------*
\item $K\subset A$.
%------------------------------------------------------------------*
\end{rlist}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%------------------------------------------------------------------*
\item Let $F$ be a field having algebraic closure $\Omega$. Let
$F\subset E\subset \Omega$ and $F\subset L\subset \Omega$. Suppose
that $E$ and $L$ are Galois extensions of $F$. Show that $E\cap L$ is
a Galois extension of $F$.
%------------------------------------------------------------------*
\item Let $F,L,E$ and $\Omega$ be as in part (c). (In particular, we
assume that $E/F$ and $L/F$ are Galois.) Considering the Galois
extension $F\subset E$, let $A(E)$ be the field $A$ discussed in part
(a). Analogously, define $A(L)$ and $A(E\cap L)$. (So, for instance,
$A(E\cap L)$ is the largest field between $F$ and $E\cap L$ such that
$A(E\cap L)/F$ is Galois and has an abelian Galois group.)

Show that $A(E\cap L)=A(E)\cap A(L)$.
%------------------------------------------------------------------*
\end{alist}
