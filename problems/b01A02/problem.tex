Let $G$ be a finite group.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Suppose that $G$ has order $n$ and $p$ is the smallest prime 
divisor of $n$. Show that any subgroup of index $p$ is normal.
%------------------------------------------------------------------*
\item Suppose that $G$ has order $255$. Prove that $G$ has a normal
subgroup of order $17$ and that $G$ has a cyclic normal subgroup of
order $85$. From these facts deduce that any group of order $255$ is
cyclic.
%------------------------------------------------------------------*
\end{alist}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\soln{The prime factorization of $255$ is $3*5*17$. The prime
factorization of $85$ is $5*17$.}
