\begin{alist}
%------------------------------------------------------------------*
\item Let $S_4$ denote the group of permutations of $4$
things. Describe the conjugacy classes in $S_4$, determine the normal
subgroups of $S_4$ and show that $S_4$ is solvable.
%------------------------------------------------------------------*
\item Suppose $G$ is a group of order $24$ which is not isomorphic to
$S_4$ and in which a Sylow $3$-subgroup is not normal. Show that
there is a homomorphism $\varphi:G\rightarrow S_4$ whose kernel is of
order $2$ and that the Sylow $2$-subgroup of $G$ is normal.
%------------------------------------------------------------------*
\item Give an explicit example of a group of the type considered in
(b).
