\begin{alist}
%------------------------------------------------------------------*
\item Let $E\leq [0,1]$ be a measurable set with Lebesgue measure $|E|>0$
and let $\chi_E$ denote the characteristic function of $E$.  Show that
the convolution $\chi_E\ast\chi_E(X)\equiv \int_{-\infty}^{\infty}
\chi_E(x-y)\chi_E(y) dy$ is a continuous function of $x$.
%------------------------------------------------------------------*
\item If $E+t=\{x+t \pipe x\in E\}$, show that there exists some
$\epsilon>0$ such that $|E\cap (E+t)|>0$ whenever $|t|<\epsilon$.
%------------------------------------------------------------------*
\end{alist}
