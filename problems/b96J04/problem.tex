Let $f(x)=x^8+1$. 
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Let $K$ be the splitting field of $f(x)$ over $\Q$
the field of rational numbers. Determine the Galois group of $K/Q$.
%------------------------------------------------------------------*
\item How many subfields of $K$ are of degree $4$ over $\Q$? How
many of these are Galois over $\Q$? Explain.
%------------------------------------------------------------------*
\item Let $L$ be the splitting field of $f(x)$ over $F_{41}$ (the field of
$41$ elements). Determine the Galois group of $L/F_{41}$.
%------------------------------------------------------------------*
\end{alist}
