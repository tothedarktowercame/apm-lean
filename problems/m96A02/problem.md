%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item State and prove Banach's contraction mapping theorem for an operator
$T:M\rightarrow M$ on a complete metric space $M$.
%------------------------------------------------------------------*
\item Use part (a) to show that if $T:M\rightarrow M$ is on opererator
for which the composition $T^N$ is a contraction for some $N\geq 1$,
then $T$ has a unique fixed point.
%------------------------------------------------------------------*
\item Suppose that a Lipschitz continuous function $f:\R\rightarrow
\R$ and a number $u_0\in \R$ are given. Use either part (a) or part
(b) above to prove that there exists a unique solution $u$ of the initial
value problem $u'(t)=f(u(t))$, ($0<t<1$), $u(0)=u_0$.
%------------------------------------------------------------------*
\end{alist}
