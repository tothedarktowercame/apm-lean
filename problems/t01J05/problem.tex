Let $X$ be a compact, oriented manifold without boundary.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item State the definition of a partition of unity subordinate to an
open cover (of $X$) $\{U_{\alpha}\}$.
%------------------------------------------------------------------*
\item Suppose that there is a nowhere-zero $1$-form $\omega$ on
$X$. Prove that the Euler characteristic of $X$ is zero.
%------------------------------------------------------------------*
\end{alist}
