\begin{alist}
%------------------------------------------------------------------*
\item Prove that the set of functions in $C[-1,1]$ that do not posses a
derivative at the point $0$ is a set of the second category. In this
space we use the topology that arises from the norm
$\|x\|=\sup_{-1\leq t\leq 1}|x(t)|$.
%------------------------------------------------------------------*
\item Let $X$ be a Banach space and let $T$ be a linear map of $X$
into $X$. Suppose that there is a continuous linear and injective map
$A:X\rightarrow X$ such that $AT$ is continuous. Prove or disprove
that $T$ then has to be continuous.
%------------------------------------------------------------------*
\end{alist}
