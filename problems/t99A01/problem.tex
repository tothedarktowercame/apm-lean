Compute the fundamental group and all homology groups of the spaces
$X$, $Y$, and $Z$ below:
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item $X$ is obtained by attaching a $2$-cell $D$ to a circle $S$ by a
map of degree $2$ from the boundary circle of $D$ to $S$.
%------------------------------------------------------------------*
\item $Y$ is obtained from $X$ by attaching another $2$-cell $D'$ by a
map of degree $4$ from the boundary circle of $D'$ to $S$.
%------------------------------------------------------------------*
\item $Z$ is obtained from $Y$ by attaching yet another $2$-cell $D''$
by a map of degree $5$ from the boundary circle of $D''$ to $S$.
%------------------------------------------------------------------*
\end{alist}
