Let $\alpha=3dx\wedge dy$ be a $2$-form on $\Rtw$. Let $\beta$ be the
induced $2$-form on $\Ttw$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Compute $\int_{\Ttw}\beta$.
%------------------------------------------------------------------*
\item Let $f:\Rtw\rightarrow \Rtw$ be the map $f(x,y)=(5
x+2y,4x+3y)$. This induces a map $g:\Ttw\rightarrow \Ttw$. Compute
$\int_{\Ttw}g^*(\beta)$.
%------------------------------------------------------------------*
\item Show that $g$ is not homotopic to the identity.
%------------------------------------------------------------------*
\end{alist}
