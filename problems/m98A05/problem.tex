\begin{alist}
%------------------------------------------------------------------*
\item State the Implicit Function theorem.
%------------------------------------------------------------------*
\item Let $X$ and $Y$ be Banach spaces. Let $F, G\in C^1(X,Y)$, and
let $H(x,\epsilon)=F(x)+\epsilon G(x)$ for $\epsilon\in \R$. If
$H(x_0,\epsilon_0)=0$ and $DF(x_0)$ is invertible, show that there
exists $x\in X$ such that $H(x,\epsilon)=0$ for $\epsilon$
sufficiently close to $\epsilon_0$.
%------------------------------------------------------------------*
\item For small $\epsilon$, prove that there is a solution $w\in
H^2(0,\pi)$ to the problem $w''=w+\epsilon w^2, w(0)=w(\pi)=0$.
%------------------------------------------------------------------*
\end{alist}
