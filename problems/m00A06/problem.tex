Let $f(x)=|x|^{-1}$ on $\R$. Consider $T_f$ defined by
$T_f(\varphi)=\int_{-\infty}^{\infty}|x|^{-1}\varphi(x)dx$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that $T_f$ is not well-defined for $\varphi\in
\mathcal{D}$.
%------------------------------------------------------------------*
\item Show that $T_f$ is defined for all $\varphi\in \mathcal{D}$
which vanish at the origin.
%------------------------------------------------------------------*
\item Show that there is a $T\in \mathcal{D}'$ such that $T$ agrees
with $T_f$ on test functions that vanish at the origin.
%------------------------------------------------------------------*
\end{alist}
