Let $X$ be normal and Hausdorff with a countable basis. Let $U$ be an
open set in $X$. 
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that $U$ is a countable union of closed sets in
$X$.
%------------------------------------------------------------------*
\item Show that there is a continuous function $f:X\rightarrow [0,1]$
such that $f(x)>0$ for $x\in U$ and $f(x)=0$ for $x\notin U$.

%------------------------------------------------------------------*
\end{alist}
