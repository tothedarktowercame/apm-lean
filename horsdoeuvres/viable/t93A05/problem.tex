\begin{alist}
%------------------------------------------------------------------*
\item Prove that every smooth map $f:\Stw\rightarrow \RP^3$ is
homotopic to a constant map.
%------------------------------------------------------------------*
\item Find a compact $1$-manifold $C$ and a compact surface $F$ in
$\RP^3$ with nonzero mod $2$ intersection number -- or, prove that
none exist.
%------------------------------------------------------------------*
\item Find \emph{oriented} submanifolds $C$ and $F$ as in (b) with
nonzero \emph{oriented} intersection number, or prove that none exist.
%------------------------------------------------------------------*
\end{alist}
