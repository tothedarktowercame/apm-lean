Let $F$ be a field of characteristic $0$ and let $K\supset F$ be a
finite field extension. All the fields in this question will be
subfields of $K$ containing $F$. If $L_1$ and $L_2$ are two such
fields, let $L_1L_2$ be the smallest subfield of $K$ containing them
both.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that as a set $L_1L_2$ consists of all sums of the form
$\sum_{i}l_{1,i}l_{2,i}$, where $l_{j,i}\in L_j$.
%------------------------------------------------------------------*
\item Now (and for the rest of the problem) $L_1$ and $L_2$ are Galois
over $F$.

Show that $L_1L_2$ is also Galois over $F$. Hint: splitting fields.
%------------------------------------------------------------------*
\item Assume $K/F$ is Galois over $F$ with group $G$ and $H_i\subset
G$ is the subgroup associated to $L_i$. Show that $H_1\cap H_2$ is the
subgroup associated to $L_1L_2$. Show that $H_1H_2$ is the subgroup
associated to $L_1\cap L_2$.
%------------------------------------------------------------------*
\item Suppose that $K=L_1L_2$ and that both $L_1/F$ and $L_2/F$ have
abelian Galois groups. Show that $K/F$ has an abelian Galois group.
%------------------------------------------------------------------*
\end{alist}
