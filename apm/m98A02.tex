Let $X$ and $Y$ be Banach spaces and let $A:X\rightarrow Y$ be a
linear operator.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Define what it means for $A$ to be bounded and prove this is
equivalent to continuity. Show that the set of all bounded operators
forms a Banach space $B(X,Y)$.
%------------------------------------------------------------------*
\item Suppose that $A\in B(X,Y)$ and $\|A\|_{B(X,Y)}<1$. Show that
$I-A$ is invertible, where $I:X\rightarrow X$ is the identity
map. Also show that $1/(1+\|A\|)\leq \|(I-A)^{-1}\|\leq 1/(1-\|A\|)$.
%------------------------------------------------------------------*
\end{alist}
