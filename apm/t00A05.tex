Let $\omega$ denote the $1$-form on $\Rth\setminus\{z-\text{axis}\}$
given by $\omega=\frac{y}{x^2+y^2}dx-\frac{x}{x^2+y^2}dy+\cos z dz$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Compute $d\omega$. (And check your work.)
%------------------------------------------------------------------*
\item Evaluate $\int_{C_1}\omega$ where $C_1$ is the unit circle in
the $(x,y)$-plane oriented counterclockwise.
%------------------------------------------------------------------*
\item Evaluate $\int_{C_2}\omega$ where $C_2$ is the counterclockwise
oriented intersection of the ellipsoid $2x^2+y^2+z^2=10$ with the
graph $z=e^{y-x}$.
%------------------------------------------------------------------*
\end{alist}
