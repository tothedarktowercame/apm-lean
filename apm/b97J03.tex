Let $P(x)$ be an irreducible polynomial in $\Q[x]$. Let $\alpha\in \C$
be a root of $P(x)$. Let $n$ be the degree of $P(x)$ and let $r$ be
the number of roots of $P(x)$ in the field $\Q(\alpha)$. Let $G$ be
the group of $P(x)$ \ie\ the Galois group of a splitting field of
$P(x)$ over $\Q$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that $r$ divides $n$.
%------------------------------------------------------------------*
\item Show that if $G$ is abelian, then $r=n$.
%------------------------------------------------------------------*
\item Determine $r$ if $P(x)=x^6-2$, giving the reasoning behind your
answer.
%------------------------------------------------------------------*
\end{alist}
