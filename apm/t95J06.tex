A vector field on $\Sfi\subset \C^3\cong \R^6$ is given by
$v(\mathbf{z})=i\mathbf{z}$. Let $H:\Sfi\times \R \rightarrow \Sfi$ be
the corresponding flow. Let $\T$ be the torus $\{(\cos \theta
e^{i\varphi},\sin \theta e^{i\varphi},0)\pipe 0\leq \theta<2 \pi,0\leq
\varphi<2\pi\}\subset \Sfi$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that $T$ is invariant under $H$, \ie\ $H(T\times \R)\subset
T$.
%------------------------------------------------------------------*
\item Suppose that a compact orientable $2$-manifold $F$ in $\Sfi$ is
invariant under $H$. List all possible diffeomorphism types of $F$. 
%------------------------------------------------------------------*
\end{alist}
