\begin{alist}
%------------------------------------------------------------------*
\item Let $T\subset \mathbf{A}^3$ be the
torus obtained by revolving the circle of radius $r$ and center
$(R,0,0)$ that lies in the $x$-$z$-plane about the $z$ axis, where
$x,y,z$ are standard coordinates on $\mathbf{A}^3$ and
$R>r>0$. Consider the differential $2$-form $\omega=x \: dy\wedge
dz-d(yx\:dz)$ on $\mathbf{A}^3$. Compute $\int_T\omega$. (Orient $T$
as the boundary of the compact region it encloses.)
%------------------------------------------------------------------*
\item Construct or prove the non-existence of a closed $1$-form $\eta$
on $\mathbf{A}^3$ and a smooth map $\gamma:\Son\rightarrow T$ such
that $\int_{\Son}\gamma^*\eta\neq 0$.
%------------------------------------------------------------------*
\end{alist}
