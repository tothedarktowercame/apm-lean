Consider a compact set $K\subset \R$ with positive Lebesgue measure.
Define the function $g$ on $\C \setminus K$ by
$g(z)=\int_K\frac{1}{t-z}dt$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Prove that $g$ is analytic on $\C \setminus K$.
%------------------------------------------------------------------*
\item Prove that $g$ cannot be extended by analytic continuation to an
entire function.
%------------------------------------------------------------------*
\item Show that $\lim_{z\rightarrow \infty}[zf(z)]$ exists and
determine its value.
%------------------------------------------------------------------*
\end{alist}
