Fourier transform.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Define the Fourier transform $\hat{f}$ of $f\in L^1(\R)$.
%------------------------------------------------------------------*
\item Express the Fourier transform of $f'$ in terms of the Fourier
transform of $f$.
%------------------------------------------------------------------*
\item Use the priciple in the preceding part of the problem to take 
the Fourier transform on the space variable in the boundary value
problem $u_{xx}=u_t$ and $u(x,0)=f(x)$. Assume that $-\infty<x<\infty$
and $t>0$.
%------------------------------------------------------------------*
\item Complete the solution of the boundary value problem obtaining
$u(x,t)$ as a convolution of suitable functions.
%------------------------------------------------------------------*
\end{alist}
