Let $G$ be a group and let $H$ be a subgroup with finite index.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item If $g$ is an element of $G$ show that exists a smallest
positive integer $k$ such that $g^k\in H$ and show that $k$ divides
every integer $m$ for which $g^m\in H$.
%------------------------------------------------------------------*
\item Show that if $H$ is normal in $G$ then the number $k$ in (a) must
divide the index of $H$ in $G$.
%------------------------------------------------------------------*
\item Show by a counterexample that the statement in (b) does not always
remain true if $H$ is not normal in $G$.
%------------------------------------------------------------------*
\end{alist}
