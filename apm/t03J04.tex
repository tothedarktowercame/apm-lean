Let $T^n=\Rn/\Z^n=\Son\times \Son \times \cdots \times \Son$ denote the
$n$-torus.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{rlist}
%------------------------------------------------------------------*
\item Give an explicit example of a vector field on $T^n$ whose zeroes
are isolated.
%------------------------------------------------------------------*
\item Compute the Euler characteristic $\chi(T^n)$.
%------------------------------------------------------------------*
\end{rlist}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%------------------------------------------------------------------*
\item Let $X$ be a smooth oriented manifold with an
orientation-preserving local diffeomorphism $\varphi:T^n\rightarrow
X$. List all possible values of $\chi(X)$ and prove your answer.
%------------------------------------------------------------------*
\end{alist}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\soln{Constant vector field.}
