Let $f(x)=|x|^{-2}(1+|x|^2)^{-1}$ for $x\in \Rth$, which we equip with
three-dimensional Lebesgue measure $m_3$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item For which values of $p$ in $(0,\infty]$ is it the case that
$f\in L^p(\Rth)$?
%------------------------------------------------------------------*
\item Compute $\int_{\Rth}f(x)dm_3$.
%------------------------------------------------------------------*
\item Verify that the convolution of $f$ with an $L^2$ function
$g$ belongs to $L^2(\Rth)$ and determine an explicit constant $c$ for
which $\|f\ast g\|_2\leq c\|g\|_2$ for all such $g$.

%------------------------------------------------------------------*
\end{alist}
