Let $X$ be a smooth manifold of dimension at least $1$. Let $\omega_p$
be a nonzero element of $T_p^*X$ for some $p\in X$. 
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Construct a smooth $1$-form $\omega$ on $X$ whose value at $p$ is 
$\omega_p$. 
%------------------------------------------------------------------*
\item Can we always construct $\omega$ in part (a) to be closed? Provide 
a proof or a counterexample.
%------------------------------------------------------------------*
\end{alist}
