Assume that $f:[0,1]\rightarrow \R$ is bounded and Lebesgue measurable
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Establish that 
$\check{\int}_0^1 f(x)dx \leq \int_0^1 f(x)dx \leq \hat{\int}_0^1
f(x)dx$, where $\check{\int}, \int,\hat{\int}$ signify the lower
Riemann integral, the Lebesgue integral, and the upper Riemann
integral respectively.
%------------------------------------------------------------------*
\item Suppose now that $f(x)=\chi_K(X)$ for almost every $x\in [0,1]$, 
where $K$ is a compact subset of $[0,1]$ with positive Lebesgue
measure and empty interior. Prove that $f$ is not Riemann integrable.
%------------------------------------------------------------------*
\end{alist}
