Let $G$ be a finite group and $p$ the smallest prime dividing the
order of $G$. Let $H$ be a subgroup with index $p$ in $G$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Prove that $H$ is a normal subgroup of $G$.
%------------------------------------------------------------------*
\item Prove that if $p+2$ is a prime and $G$ is of order $p(p+2)$ 
then $G$ is cyclic.
%------------------------------------------------------------------*
\end{alist}
