Let $f\in L^1(\R,m)$ and $g\in L^p(\R,m)$ where $1<p<\infty$ and $m$
denotes Lebesgue measure.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that for $m$ almost all real $x$ the function defined by
$y\mapsto f(x-y)g(y)$ is integrable on $\R$ with respect to $m$.
%------------------------------------------------------------------*
\item Define a function $h$ at $m$-almost all real $x$ by
$h(x)=\int_{\R}f(x-y)g(y)dm(y)$. Show that $\|h\|_p\leq
\|f\|_1\|g\|_p$.
%------------------------------------------------------------------*
\end{alist}
