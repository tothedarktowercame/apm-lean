True or false with proof:
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Every three dimensional manifold is orientable.
%------------------------------------------------------------------*
\item If $X$ is a compact oriented manifold and $f:X\rightarrow X$ is
a diffeomorphism, then $f$ has Lefschetz number equal to $1$.
%------------------------------------------------------------------*
\item Let $X$ and $Y$ be compact connected oriented manifolds of the
same dimension. Suppose $f:X\rightarrow Y$ is smooth and $y_0,y_1\in
Y$ are regular values. Then the cardinalities of the sets
$f^{-1}(y_0)$ and $f^{-1}(y_1)$ are equal.
%------------------------------------------------------------------*
\end{alist}
