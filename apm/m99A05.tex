Suppose that $X$ is a normed linear space and that
$\{u_n\}_{n=1}^{\infty}$ is a sequence in $X$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show by example if $u_n$ converges weakly to $u$, then $u_n$
need not converge strongly to $u$.
%------------------------------------------------------------------*
\item Prove that $u_n\rightarrow u$ does imply that $u_n$ converges
weakly to $u$.
%------------------------------------------------------------------*
\item If $\dim X<\infty$, prove that $u_n\rightarrow u$ if and only if
$u_n$ converges weakly to $u$.
%------------------------------------------------------------------*
\item Prove that if $u_n$ converges weakly to $u$, then $\|u_n\|$ is
bounded independently of $n$.
%------------------------------------------------------------------*
\item Prove that weak limits are unique.
%------------------------------------------------------------------*
\end{alist}
