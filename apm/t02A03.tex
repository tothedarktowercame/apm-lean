\begin{alist}
%------------------------------------------------------------------*
\item Let $K$ be the simple closed curve $\{0\}\times \Son$ in the
solid torus $D^2\times \Son$ and let $X=D^2\times \Son \setminus K$.
Compute $\pi_1(X)$.
%------------------------------------------------------------------*
\item Let $W$ be the bounded region in 3-space whose boundary is a
standardly embedded genus two surface and let $C\subset
\mathrm{int}(W)$ be the simple closed curve going around one of the
holes in the same way as $K$ went around one of the holes in the
previous part. Set $Y=W \setminus C$. Compute $\pi_1(Y)$ and $H_n(Y)$,
$n\geq 0$.
%------------------------------------------------------------------*
\end{alist}
