Let $K$ be the Klein bottle.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that $\pi_1(K)\cong \langle a,b \pipe
b^{-1}ab=a^{-1}\rangle$.
%------------------------------------------------------------------*
\item Let $A$ be the subgroup $\{a^n \pipe n\in \Z\}$ of
$\pi_1(K)$. Let $X$ be a path connected, locally path connected,
semilocally simply connected, subspace of $K$ such that $\im
i_{*}\subset A$ where $i_{*}:\pi_1(X)\rightarrow \pi_1(K)$ is the map
induced by inclusion. Show that $X$ embeds in the plane $\Rtw$.
%------------------------------------------------------------------*
\end{alist}
