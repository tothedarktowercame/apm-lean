\begin{alist}
%------------------------------------------------------------------*
\item State the Fundamental Theorem of Finite Abelian Groups. (\Ie\ 
the theorem that describes all possible structures for finite Abelian
groups.)
%------------------------------------------------------------------*
\item Give necessary and sufficient conditions for $\Z/m\Z\oplus
\Z/n\Z$ to be a cyclic group ($m$ and $n$ are positive integers).
%------------------------------------------------------------------*
\item Let $K$ be a field and let $G\subset K^*$ be a finite
multiplicative subgroup of the units in $K$. Prove that $G$ is cyclic.
Hint: consider polynomials of the form $x^i-1$ for some $i$.
%------------------------------------------------------------------*
\end{alist}
