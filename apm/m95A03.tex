\begin{alist}
%------------------------------------------------------------------*
\item Explain with as much detail as possible the equivalence between these
two problems: (i) $(pu')'-qu=f$ ($a\leq t\leq b$),
$u(a)=\alpha,u(b)=\beta$; and, (ii) minimize $\int_a^b[p(u')^2+q u^2+2
u f] dt$ subject to the constraint $u(a)=\alpha, u(b)=\beta$.
%------------------------------------------------------------------*
\item Show how to change problem (i) into an equivalent one having
homogeneous boundary conditions.
%------------------------------------------------------------------*
\item With appropriate assumptions on $p(x)$, $q(x)$ prove existence
and uniqueness of a solution to (ii) when $\alpha=\beta=0$.
%------------------------------------------------------------------*
\end{alist}
