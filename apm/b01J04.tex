\begin{alist}
%------------------------------------------------------------------*
\item Let $\C^{n\times n}$ denote the set of $n\times n$ matrices with
entries in $\C$ and let $A,B\in \C^{n\times n}$. Prove that there
exists a nonsingular $T\in \C^{n\times n}$ such that $TAT^{-1}$ and
$TBT^{-1}$ are diagonal $\Leftrightarrow AB=BA$.
%------------------------------------------------------------------*
\item Let $k$ be a field and let $A\in k^{n\times n}$ be
nonsingular. Prove that there exists a polynomial $q(x)\in k[x]$ such
that $A^{-1}=q(A)$.
%------------------------------------------------------------------*
\end{alist}
