Let $F_q$ denote a finite field with $q$ elements. Consider the field
$F_q(x)$ of rational functions with coefficients in $F_q$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that for any $a,b\in F_q$ with $a\neq 0$, there exists a unique
isomorphism $\sigma_{a,b}$ of $F_q(x)$ with $\sigma_{a,b}(x)=ax+b$ and
$\sigma_{a,b}(c)=c$ for all $c\in F_q$.
%------------------------------------------------------------------*
\item Show that the set $G=\{\sigma_{a,b}\pipe a,b\in F_q, a\neq 0\}$
is a group of order $q(q-1)$.
%------------------------------------------------------------------*
\item Show that the fixed field of $G$ in $F_q(x)$ is
$F_q((x^q-x)^{q-1})$.
%------------------------------------------------------------------*
\item Let $K$ be a subfield of $F_q(x)$ which properly contains
$F_q$. Show that $F_q(x)$ is finite over $K$.
%------------------------------------------------------------------*
\end{alist}
