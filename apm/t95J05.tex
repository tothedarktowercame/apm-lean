Let $\omega=(x_2^3-4x_1)dx_2\wedge dx_3\wedge dx_4\in \Omega^3(\Rfo)$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Prove that there is a unique form $\eta\in \Omega^3(\RP^3)$ such
that $\pi^*\eta=i^*\omega$, where $i:\Sth\rightarrow \Rfo$ is
inclusion and $\pi:\Sth\rightarrow \RP^3$ is the usual projection.
%------------------------------------------------------------------*
\item Compute $\int_{\RP^3}\eta$. Hint: $\vol \mathbf{B}^4=\pi^4/2$.
%------------------------------------------------------------------*
\end{alist}
