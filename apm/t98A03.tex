Consider the space $X=T^2\vee \RP^2$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Is $X$ a manifold?
%------------------------------------------------------------------*
\item Compute the fundamental group of $X$.
%------------------------------------------------------------------*
\item Construct a regular $3$-fold connected covering of $X$.
%------------------------------------------------------------------*
\item Does $X$ have irregular $3$-fold coverings. If so, construct
one. If not, prove it.
%------------------------------------------------------------------*
\end{alist}
