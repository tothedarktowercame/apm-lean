Let $\Stw$ be the unit $2$-sphere in $\Rth$ and $f:\Stw\rightarrow
\Stw$ a rotation through $\pi$ in the $z$-axis. Let $\Son$ be the unit
circle in $\Rtw$ and $g:\Son\rightarrow \Son$ reflection through the
$y$-axis. Let $h:\Stw\times \Son\times \Son\rightarrow \Stw\times
\Son\times \Son$ be defined by $h(p,q,r)=(f(p),g(q),g(r))$. 
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Compute the Lefschetz number of $h$.
%------------------------------------------------------------------*
\item Is $h$ homotopic to the identity?
%------------------------------------------------------------------*
\end{alist}
