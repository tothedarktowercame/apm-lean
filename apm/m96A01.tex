\begin{alist}
%------------------------------------------------------------------*
\item Define the term ``compact linear transformation''.
%------------------------------------------------------------------*
\item Prove that if $L$ is a compact linear transformation of a normed
linear space $X$ into itself, and $\lambda$ is a nonzero complex
number, then the subspace $\{x\in X \pipe Lx=\lambda x\}$ is finite
dimensional.
%------------------------------------------------------------------*
\end{alist}
