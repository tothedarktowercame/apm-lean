Let $K\subset F$ be fields with $F$ a finite dimensional Galois
extension of $K$ (\ie\ $F$ is a normal and separable extension). Let
$p\neq q$ be prime integers and assume that $[F:K]=p^nq^m$ with
$n,m>1$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that there is a field $L$ with $K\subset L\subset F$ and
with $[F:L]=p^n$.
%------------------------------------------------------------------*
\item Let $L$ be as in part (a).

Show that $\{L' \pipe L' \text{\ is a field with\ } K\subset L'\subset
F \text{\ and\ } [F:L']=p^n\}=\{L\}$ if and only if $L$ is a Galois
extension of $L$.
%------------------------------------------------------------------*
\end{alist}
