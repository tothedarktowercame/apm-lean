Let $B_{m,1}$ denote the ball of radius $1$ centered at the origin in
$\Rm$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{alist}
%------------------------------------------------------------------*
\item Show that $\exists f:\R\rightarrow [0,1]$ such that
$m\{B_{n+1,1}\}=m\{B_{n,1}\}\int [f(t)]^n dt$.
%------------------------------------------------------------------*
\item Show that $\int[f(t)]^n dt\rightarrow 0$ as $n\rightarrow
\infty$.
%------------------------------------------------------------------*
\item Show that for any finite positive number $A$, $A^n
m\{B_{n,1}\}\rightarrow 0$ as $n\rightarrow \infty$.
%------------------------------------------------------------------*
\end{alist}
